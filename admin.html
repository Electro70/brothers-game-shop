import { auth, onAuthStateChanged, signInWithEmailAndPassword, signOut } from "./firebase.js"; const BLOCKED_ADMIN_EMAILS = [ "theummah234@gmail.com" ]; function isBlockedAdmin(email) { if (!email) return false; return BLOCKED_ADMIN_EMAILS.includes(email.trim().toLowerCase()); } // যদি আগে থেকেই অ্যাডমিন জিমেইল লগইন থাকে, সাথে সাথে সাইন আউট করবে onAuthStateChanged(auth, async (user) => { if (user && isBlockedAdmin(user.email)) { await signOut(auth); alert("অ্যাডমিন জিমেইল দিয়ে ইউজার ওয়েবসাইটে লগইন করা যাবে না।"); return; } if (user) { window.location.href = "index.html"; } }); // লগইন সাবমিটের সময় অ্যাডমিন জিমেইল ব্লক করা async function handleUserLogin(email, password) { const cleanEmail = email.trim().toLowerCase(); if (isBlockedAdmin(cleanEmail)) { alert("অ্যাডমিন জিমেইল দিয়ে ইউজার ওয়েবসাইটে লগইন করা যাবে না।"); return; } const cred = await signInWithEmailAndPassword(auth, cleanEmail, password); if (isBlockedAdmin(cred.user.email)) { await signOut(auth); alert("অ্যাডমিন জিমেইল দিয়ে ইউজার ওয়েবসাইটে লগইন করা যাবে না।"); return; } window.location.href = "index.html"; }
